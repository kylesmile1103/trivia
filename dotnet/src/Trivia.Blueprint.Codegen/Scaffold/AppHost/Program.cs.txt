using Microsoft.Extensions.Configuration;

var builder = DistributedApplication.CreateBuilder(args);

var postgres = builder.AddPostgres("postgres");
if (builder.Configuration.GetValue("Trivia:UseDataVolume", true))
    postgres.WithDataVolume();
var db = postgres.AddDatabase("appdb");

var migrations = builder.AddProject<Projects.{{App}}_MigrationService>("migrations")
    .WithReference(db, connectionName: "AppDb")
    .WaitFor(db);

var api = builder.AddProject<Projects.{{App}}_Api>("api", options => options.ExcludeLaunchProfile = true)
    .WithHttpEndpoint(env: "ASPNETCORE_HTTP_PORTS")
    .WithReference(db, connectionName: "AppDb")
    .WaitForCompletion(migrations)
    .WithHttpHealthCheck("/health");

if (builder.Configuration.GetValue("Trivia:StartWeb", true))
{
    builder.{{WebResource}}("web", "../../web")
        .WithPnpm()
        .WithReference(api)
        .WaitFor(api);
}

builder.Build().Run();
