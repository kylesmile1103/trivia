using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using {{Namespace}}.Data;
using {{App}}.MigrationService;

HostApplicationBuilder builder = Host.CreateApplicationBuilder(args);
builder.AddServiceDefaults();
string connection = builder.Configuration.GetConnectionString("AppDb")
    ?? throw new InvalidOperationException("Set ConnectionStrings__AppDb to run the migration service.");
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseNpgsql(connection, postgres => postgres.EnableRetryOnFailure()));
builder.Services.AddHostedService<Worker>();

await builder.Build().RunAsync();
