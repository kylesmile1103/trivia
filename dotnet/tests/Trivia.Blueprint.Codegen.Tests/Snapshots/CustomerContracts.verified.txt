{

    [JsonPropertyName("creditLimit"), Range(typeof(decimal), "0", "1000000", ParseLimitsInInvariantCulture = true, ConvertValueInInvariantCulture = true)]

    [JsonPropertyName("birthDate")]
    public DateOnly? BirthDate { get; init; }

    [JsonPropertyName("externalRef")]
    public Guid? ExternalRef { get; init; }
}
